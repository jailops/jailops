// gen.go.txt is the generator of one upgrade fixture. scripts/make-upgrade-fixture.sh copies it into a checkout of a
// RELEASE TAG as cmd/upgrade-fixture-gen/main.go and runs it there, so the files it writes are written by the code of
// that release: the state store writes its objects and migrates its database, lifecycle writes the records of the
// jails, the daemon yields its default configuration. The specs below are written as JSON for the schema of that
// release (the `old` variants carry the fields that were removed later) and are decoded STRICTLY with the types of
// the release, so a spec this file gets wrong for a release fails here and not silently in the fixture.
//
// It only uses what every release since 0.8.0 has: internal/state, internal/store, internal/lifecycle (Record and
// Bundle), internal/daemon (Defaults), sdk/api.
package main

import (
	"bytes"
	"context"
	"encoding/json"
	"flag"
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strconv"
	"strings"
	"time"

	"gopkg.in/yaml.v3"

	"jailops.net/jailops/internal/daemon"
	"jailops.net/jailops/internal/lifecycle"
	"jailops.net/jailops/internal/state"
	"jailops.net/jailops/internal/store"
	"jailops.net/jailops/sdk/api"
)

const (
	hostUUID = "00000000-0000-4000-8000-0000000000a1"
	project  = "shop"
)

var stamp = time.Date(2026, 1, 2, 3, 4, 5, 0, time.UTC)

type doc struct {
	kind    api.Kind
	name    string
	uuid    string
	project string
	scope   api.Scope
	spec    string
	typed   any // a pointer to the typed spec of the release, filled by decoding spec strictly
}

func must(err error, what string) {
	if err != nil {
		fmt.Fprintf(os.Stderr, "upgrade-fixture-gen: %s: %v\n", what, err)
		os.Exit(1)
	}
}

func strict(s string, v any) error {
	dec := json.NewDecoder(strings.NewReader(s))
	dec.DisallowUnknownFields()
	return dec.Decode(v)
}

// oldSchema reports whether the release still has the fields 0.8.4 removed: ipv6.mode, jail proxy devices, redirect
// rules of policies, image aliases and wireguard `down`.
func oldSchema(version string) bool {
	v := strings.TrimPrefix(version, "v")
	parts := strings.Split(v, ".")
	if len(parts) != 3 {
		must(fmt.Errorf("version %q is not X.Y.Z", version), "version")
	}
	minor, _ := strconv.Atoi(parts[1])
	patch, _ := strconv.Atoi(parts[2])
	return minor < 8 || (minor == 8 && patch <= 3)
}

func objects(old bool) []doc {
	lan6 := `{"ndp_proxy":true,"subnet":"2001:db8:77:1::/64","range":"2001:db8:77:1::10-2001:db8:77:1::ff"}`
	routed6 := `{"subnet":"2001:db8:78:1::/64","range":"2001:db8:78:1::10-2001:db8:78:1::ff"}`
	imageSpec := `{"os":"freebsd","release":"15.1-RELEASE","arch":"amd64","source":{"type":"pkgbase"},"retention":{"keep_last":3}}`
	wg := `{"mode":"server","address":["10.99.0.1/24"],"listen_port":51820,"networks":["lan"],"disabled":true}`
	if old {
		lan6 = `{"mode":"proxy","subnet":"2001:db8:77:1::/64","range":"2001:db8:77:1::10-2001:db8:77:1::ff"}`
		routed6 = `{"mode":"routed","subnet":"2001:db8:78:1::/64","range":"2001:db8:78:1::10-2001:db8:78:1::ff"}`
		imageSpec = `{"os":"freebsd","release":"15.1-RELEASE","arch":"amd64","source":{"type":"pkgbase"},"aliases":["stable"],"retention":{"keep_last":3}}`
		wg = `{"mode":"server","address":["10.99.0.1/24"],"listen_port":51820,"networks":["lan"],"down":true}`
	}
	ipv4 := `{"subnet":"10.77.0.0/24","gateway":"10.77.0.1","range":"10.77.0.2-10.77.0.254","nat":true}`
	// The policy of the fixture: defaults stated (every release accepted that), a rule from an address set and, on
	// the old schema, a rule that carried a port forward (a redirect).
	rules := `[{"name":"web","action":"allow","direction":"in","proto":"tcp","from":["set:office"],"ports":["443"]}]`
	if old {
		rules = `[{"name":"web","action":"allow","direction":"in","proto":"tcp","from":["set:office"],"ports":["443"]},` +
			`{"name":"fwd","action":"allow","direction":"in","proto":"tcp","ports":["8080"],"redirect":{"host_port":8080,"jail_port":80}}]`
	}
	return []doc{
		{kind: api.KindStorage, name: "default", scope: api.ScopeHost, spec: `{"dataset":"zroot/jailops","compression":"lz4"}`, typed: new(api.StorageSpec)},
		{kind: api.KindNetwork, name: "lan", scope: api.ScopeFleet, spec: `{"comment":"fixture","uplink":"vtnet0","ipv4":` + ipv4 + `,"ipv6":` + lan6 + `}`, typed: new(api.NetworkSpec)},
		{kind: api.KindNetwork, name: "routed", scope: api.ScopeFleet, spec: `{"uplink":"vtnet0","ipv4":{"subnet":"10.78.0.0/24","gateway":"10.78.0.1","range":"10.78.0.2-10.78.0.254"},"ipv6":` + routed6 + `}`, typed: new(api.NetworkSpec)},
		{kind: api.KindImage, name: "freebsd-15-1", scope: api.ScopeFleet, spec: imageSpec, typed: new(api.ImageSpec)},
		{kind: api.KindProject, name: project, scope: api.ScopeHost, spec: `{"comment":"fixture","networks":[{"network":"lan"}],"storage":[{"storage":"default","quota":"50G"}],"quotas":{"jails":10,"addresses":20},"firewall_policies":["floor"],"images":["freebsd-15-1"]}`, typed: new(api.ProjectSpec)},
		{kind: api.KindFirewallPolicy, name: "floor", project: project, scope: api.ScopeFleet, spec: `{"default_in":"deny","default_out":"allow","rules":[{"action":"allow","direction":"in","proto":"icmp","from":["any"]},{"action":"allow","direction":"in","proto":"icmp6","from":["any"]}]}`, typed: new(api.FirewallPolicySpec)},
		{kind: api.KindFirewallPolicy, name: "web-access", project: project, scope: api.ScopeFleet, spec: `{"default_in":"allow","default_out":"allow","rules":` + rules + `}`, typed: new(api.FirewallPolicySpec)},
		{kind: api.KindAddressSet, name: "office", project: project, scope: api.ScopeFleet, spec: `{"addresses":[{"address":"192.0.2.0/24","comment":"office"},{"address":"2001:db8:ff::/48"}]}`, typed: new(api.AddressSetSpec)},
		{kind: api.KindVolume, name: "data", project: project, scope: api.ScopeHost, spec: `{"storage":"default","size":"10G","uid":1001,"gid":1001,"mode":"0750"}`, typed: new(api.VolumeSpec)},
		{kind: api.KindWireGuard, name: "vpn", scope: api.ScopeFleet, spec: wg, typed: new(api.WireGuardSpec)},
	}
}

type jailDoc struct {
	name, uuid, spec string
}

func jails(old bool) []jailDoc {
	webImage := `"freebsd-15-1"`
	reach := `"expose":[{"proto":"tcp","ports":["80","443"],"families":["4","6"],"comment":"the site"}]`
	if old {
		webImage = `"freebsd-15-1@stable"`
		reach = `"proxy":[{"name":"http","listen":"0.0.0.0:8080","jail_port":80,"proto":"tcp"}]`
	}
	return []jailDoc{
		{name: "web", uuid: "11111111-1111-4111-8111-111111111111", spec: `{"image":` + webImage + `,"storage":"default","layout":"thin",` +
			`"network":[{"network":"lan","ipv4":"10.77.0.2","ipv6":"2001:db8:77:1::10"}],"firewall":{"policies":["web-access"]},` +
			`"volumes":[{"volume":"data","mount":"/srv/data"}],"boot":{"autostart":true,"priority":10},"resources":{"memory":"1G","pids":200},` + reach + `}`},
		{name: "db", uuid: "22222222-2222-4222-8222-222222222222", spec: `{"image":"freebsd-15-1","storage":"default","layout":"thin",` +
			`"network":[{"network":"lan","ipv4":"10.77.0.3"}],"firewall":{"policies":["floor"]},"boot":{"autostart":true,"depends":["web"]}}`},
	}
}

func main() {
	version := flag.String("version", "", "release tag, v0.8.3")
	out := flag.String("out", "", "output directory (created)")
	flag.Parse()
	if *version == "" || *out == "" {
		must(fmt.Errorf("-version and -out are required"), "usage")
	}
	old := oldSchema(*version)
	ctx := context.Background()
	must(os.MkdirAll(*out, 0o755), "mkdir")

	// 1. the object store, written by the state package of the release
	st, err := state.Open(filepath.Join(*out, "state"), state.Options{})
	must(err, "state.Open")
	expect := map[string]map[string]string{} // kind -> name -> uuid
	for i, d := range objects(old) {
		must(strict(d.spec, d.typed), fmt.Sprintf("the %s %q does not decode strictly with the types of %s", d.kind, d.name, *version))
		u := fmt.Sprintf("00000000-0000-4000-8000-%012d", i+1)
		host := ""
		if d.scope == api.ScopeHost {
			host = hostUUID
		}
		_, err := st.Apply(state.Command{Version: state.CommandVersion, ID: fmt.Sprintf("fixture-%d", i), Time: stamp, Op: state.OpPut,
			Kind: d.kind, UUID: u, Name: d.name, Scope: d.scope, Host: host, Project: d.project,
			Spec: json.RawMessage(d.spec), ExpectGeneration: 0})
		must(err, "apply "+string(d.kind)+" "+d.name)
		if expect[string(d.kind)] == nil {
			expect[string(d.kind)] = map[string]string{}
		}
		expect[string(d.kind)][d.name] = u
	}
	must(st.Close(), "state.Close")

	// 2. the records of the jails: spec.json and bundle.json as lifecycle writes them
	var jailNames []string
	for _, j := range jails(old) {
		var spec api.JailSpec
		must(strict(j.spec, &spec), fmt.Sprintf("the jail %q does not decode strictly with the types of %s", j.name, *version))
		meta := api.ObjectMeta{Name: j.name, UUID: j.uuid, Project: project, Host: hostUUID, Scope: api.ScopeHost, Generation: 1}
		rec := lifecycle.Record{Version: 1, Metadata: meta, Spec: spec}
		bun := lifecycle.Bundle{Version: 1, Metadata: meta, Spec: spec, Volumes: spec.Volumes, Written: stamp,
			Image: lifecycle.BundleImage{Ref: "freebsd-15-1", Name: "freebsd-15-1#g1", Generation: 1, Manifest: "manifest.json", Dataset: "zroot/jailops/images/freebsd-15-1/g1"}}
		dir := filepath.Join(*out, "jails", project, j.name)
		must(os.MkdirAll(dir, 0o755), "mkdir jail")
		for name, v := range map[string]any{"spec.json": rec, "bundle.json": bun} {
			b, err := json.MarshalIndent(v, "", "  ")
			must(err, "marshal "+name)
			must(os.WriteFile(filepath.Join(dir, name), append(b, '\n'), 0o644), "write "+name)
		}
		jailNames = append(jailNames, j.name)
	}

	// 3. the database, migrated and filled by the store of the release
	dbdir := filepath.Join(*out, "data")
	must(os.MkdirAll(dbdir, 0o755), "mkdir data")
	db, err := store.Open(filepath.Join(dbdir, "jailops.db"), store.Options{})
	must(err, "store.Open")
	must(db.Users.Create(ctx, store.User{ID: "user-admin", Name: "admin", PasswordHash: "$argon2id$fixture", Permission: api.PermAdmin,
		RecoveryHashes: []string{}, Projects: []string{project}, Global: true, CreatedAt: stamp, UpdatedAt: stamp}), "user")
	must(db.Tokens.Create(ctx, store.Token{ID: "token-ci", Name: "ci", Hash: bytes.Repeat([]byte{7}, 32), Permission: api.PermOperator,
		AllowFrom: []string{}, Projects: []string{project}, CreatedAt: stamp}), "token")
	_, err = db.Audit.Append(ctx, store.AuditEntry{Time: stamp, Actor: "admin", ActorKind: "user", Method: "POST", Route: "/api/v1/jails", Object: "web", Result: "ok", Status: 202, Project: project})
	must(err, "audit")
	must(db.Meta.Set(ctx, "host_uuid", hostUUID), "meta")
	ver, err := db.Version(ctx)
	must(err, "version")
	must(db.Close(), "store.Close")

	// 4. the configuration keys of the daemon: every key the release knows (the daemon refuses unknown keys)
	cfgdir := filepath.Join(*out, "config")
	must(os.MkdirAll(cfgdir, 0o755), "mkdir config")
	yb, err := yaml.Marshal(daemon.Defaults())
	must(err, "marshal defaults")
	var tree map[string]any
	must(yaml.Unmarshal(yb, &tree), "unmarshal defaults")
	var keys []string
	var walk func(prefix string, v any)
	walk = func(prefix string, v any) {
		m, ok := v.(map[string]any)
		if !ok {
			return
		}
		for k, c := range m {
			p := k
			if prefix != "" {
				p = prefix + "." + k
			}
			keys = append(keys, p)
			walk(p, c)
		}
	}
	walk("", tree)
	sort.Strings(keys)
	must(os.WriteFile(filepath.Join(cfgdir, "jailopsd-keys.txt"), []byte(strings.Join(keys, "\n")+"\n"), 0o644), "write keys")

	// 5. what the current code must find
	exp, err := json.MarshalIndent(map[string]any{
		"version":        *version,
		"old_schema":     old,
		"schema_version": ver,
		"objects":        expect,
		"jails":          jailNames,
		"project":        project,
	}, "", "  ")
	must(err, "marshal expect")
	must(os.WriteFile(filepath.Join(*out, "expect.json"), append(exp, '\n'), 0o644), "write expect")
}
